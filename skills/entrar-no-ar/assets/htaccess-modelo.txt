# ============================================================
# Modelo de .htaccess para pagina estatica em servidor Apache.
# Vale em hospedagem compartilhada (cPanel, Hostinger e afins).
# Na Cloudways, o nginx serve css/js SEM passar por aqui: por
# isso o carimbo ?v=<timestamp> nos links continua obrigatorio.
# ============================================================

# --- Enquanto e rascunho. Remover no dia do lancamento. ---
# Lembre que a meta tag no HTML e um segundo lugar: tire os dois.
<IfModule mod_headers.c>
  Header set X-Robots-Tag "noindex, nofollow"
</IfModule>

# --- Forcar https ---
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} !=on
  RewriteRule ^(.*)$ https://seudominio.com.br/$1 [R=301,L]
</IfModule>

# --- Cache por tipo de arquivo ---
# Midia e fonte podem ser imutaveis: o nome muda quando o conteudo muda.
# css e js NAO: mesmo nome, conteudo novo a cada envio.
<IfModule mod_headers.c>
  <FilesMatch "\.(jpg|jpeg|png|gif|webp|avif|svg|mp4|webm|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>

  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=0, must-revalidate"
  </FilesMatch>

  # HTML nunca em cache: e o arquivo que aponta pra todos os outros.
  # Em servidor com proxy na frente, isso e o que evita ficar horas
  # servindo a versao velha.
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>

# --- Compressao ---
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain
  AddOutputFilterByType DEFLATE application/javascript application/json
  AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>

# --- Tipos que servidor antigo nao conhece ---
<IfModule mod_mime.c>
  AddType image/avif .avif
  AddType image/webp .webp
  AddType font/woff2 .woff2
</IfModule>

# --- Video precisa aceitar range, senao nao da pra pular no meio ---
<IfModule mod_headers.c>
  <FilesMatch "\.(mp4|webm)$">
    Header set Accept-Ranges bytes
  </FilesMatch>
</IfModule>
